%% ============================================================
% 1. TERRAIN (spatial grid - unchanged, needed for the risk map)
%% ============================================================

gridSize = 100;

[x,y] = meshgrid(1:gridSize,1:gridSize);


terrain = ...
    150 ...
    - 0.8*x ...
    + 10*exp( ...
    -((x-70).^2 + ...
    (y-50).^2)/500);


%% ============================================================
% 2. TERRAIN GRADIENT
%% ============================================================

[dzdx,dzdy] = gradient(terrain);


slopeRadians = ...
    atan( ...
    sqrt( ...
    dzdx.^2 + ...
    dzdy.^2));


slopeDegrees = ...
    rad2deg(slopeRadians);


slopeRadiansSafe = ...
    max( ...
    slopeRadians, ...
    deg2rad(0.5));


%% ============================================================
% 3. DATA COLLECTION - TERRAIN POINT DATA
%    (Sreenanda's generateSyntheticTerrain feeds real soil cohesion
%     into the grid-based stability model below)
%% ============================================================

pointTerrainData = generateSyntheticTerrain(1);

cohesion = pointTerrainData.SoilCohesion_kPa;   % kPa - from data collection

fprintf('Data Collection: generated soil cohesion = %.2f kPa (reference slope angle = %.2f deg)\n', ...
    cohesion, pointTerrainData.SlopeAngle_deg);


%% ============================================================
% 4. SOIL PARAMETERS
%% ============================================================

frictionAngle = 30;        % degrees

soilDensity = 1800;        % kg/m^3

soilDepth = 5;             % m

gamma = ...
    soilDensity * 9.81 / 1000;

frictionRadians = ...
    deg2rad(frictionAngle);


%% ============================================================
% 5. DRY FACTOR OF SAFETY
%% ============================================================

FoS = ...
    ( ...
    cohesion ...
    + gamma*soilDepth .* ...
    cos(slopeRadiansSafe).^2 .* ...
    tan(frictionRadians) ...
    ) ...
    ./ ...
    ( ...
    gamma*soilDepth .* ...
    sin(slopeRadiansSafe) .* ...
    cos(slopeRadiansSafe) ...
    );


%% ============================================================
% 6. RAINFALL PARAMETERS
%    Timeline is now DAY-based (30 days) instead of 8 hours,
%    since Data Collection generates a 30-day rainfall series.

% 8. DATA COLLECTION - SOIL MOISTURE (point-level, cross-check)

%% ============================================================
% 11. FUNCTION GENERATION - POINT-LEVEL RISK CROSS-CHECK + CSV HANDOFF

%     Generation: an independent point-based risk score, saved as a
%     CSV for the team, alongside the grid-based dashboard above.
%% ============================================================

% FUNCTION: CREATE RAINFALL EVENT (used by interactive "Apply Rainfall")

    createRainfallEvent(intensity)


rainfall = ...
    [ ...
    0 ...
    intensity*0.10 ...
    intensity*0.20 ...
    intensity*0.30 ...
    intensity*0.40 ...
    intensity*0.60 ...
    intensity*0.80 ...
    intensity ...
    ];

end



%% ============================================================
% FUNCTION: CALCULATE SIMULATION
%% ============================================================

function [ ...
    FoS_time, ...
    soilSaturation, ...
    porePressure, ...
    minimumFoS] = ...
    calculateSimulation( ...
    rainfall, ...
    slopeRadiansSafe, ...
    cohesion, ...
    frictionRadians, ...
    gamma, ...
    soilDepth, ...
    gammaWater, ...
    soilWaterCapacity, ...
    infiltrationFraction)


%% ------------------------------------------------------------
% INFILTRATED RAIN
%% ------------------------------------------------------------

infiltratedRain = ...
    rainfall .* ...
    infiltrationFraction;


%% ------------------------------------------------------------
% SOIL WATER
%% ------------------------------------------------------------

soilWater = ...
    zeros(size(rainfall));


for t = 1:length(rainfall)

    if t == 1

        soilWater(t) = ...
            min( ...
            infiltratedRain(t), ...
            soilWaterCapacity);

    else

        soilWater(t) = ...
            min( ...
            soilWater(t-1) ...
            + infiltratedRain(t), ...
            soilWaterCapacity);

    end

end


%% ------------------------------------------------------------
% SATURATION
%% ------------------------------------------------------------

soilSaturation = ...
    soilWater ./ ...
    soilWaterCapacity;


soilSaturation = ...
    min(soilSaturation,1);


%% ------------------------------------------------------------
% PORE PRESSURE
%% ------------------------------------------------------------

porePressure = ...
    soilSaturation .* ...
    gammaWater .* ...
    soilDepth;


%% ------------------------------------------------------------
% FoS
%% ------------------------------------------------------------

numRows = ...
    size(slopeRadiansSafe,1);

numCols = ...
    size(slopeRadiansSafe,2);


FoS_time = ...
    zeros( ...
    numRows, ...
    numCols, ...
    length(rainfall));


for t = 1:length(rainfall)

    u = ...
        soilSaturation(t) ...
        .* gammaWater ...
        .* soilDepth ...
        .* cos(slopeRadiansSafe).^2;


    effectiveNormalStress = ...
        gamma .* soilDepth ...
        .* cos(slopeRadiansSafe).^2 ...
        - u;


    FoS_time(:,:,t) = ...
        ( ...
        cohesion ...
        + effectiveNormalStress ...
        .* tan(frictionRadians) ...
        ) ...
        ./ ...
        ( ...
        gamma .* soilDepth ...
        .* sin(slopeRadiansSafe) ...
        .* cos(slopeRadiansSafe) ...
        );

end

%% ------------------------------------------------------------

    zeros(size(rainfall));


for t = 1:length(rainfall)

    minimumFoS(t) = ...
        min( ...
        FoS_time(:,:,t), ...
        [], ...
        "all");

end

end

function rainfallTable = generateSyntheticRainfall(numDays, seed)
    rng(seed);
    startDate = datetime('today') - days(numDays-1);
    dates = (startDate:datetime('today'))';

    baseRain = abs(5*randn(numDays,1));
    burstMask = rand(numDays,1) < 0.15; % 15% of days are "storm days"
    burstRain = burstMask .* (40 + 30*rand(numDays,1));
    rainfall = round(baseRain + burstRain, 1);

    rainfallTable = table(dates, rainfall, 'VariableNames', {'Date','Rainfall_mm'});
end

function terrain = generateSyntheticTerrain(numPoints)
    slopeAngle = 15 + 35*rand(numPoints,1); % 15-50 degrees
    cohesion = 5 + 20*rand(numPoints,1);    % kPa
    terrain = table(slopeAngle, cohesion, 'VariableNames', {'SlopeAngle_deg','SoilCohesion_kPa'});
end

function soilMoisture = generateSyntheticSoilMoisture(numDays, rainfallTable)
    decayRate = 0.85;
    moisture = zeros(numDays,1);
    moisture(1) = 20; % starting point

    for i = 2:numDays
        moisture(i) = decayRate * moisture(i-1) + 0.6*rainfallTable.Rainfall_mm(i);
    end

    soilMoisture = table(rainfallTable.Date, round(moisture,1), ...
        'VariableNames', {'Date','SoilMoisture_pct'});
end

function catalog = loadLandslideCatalog(filepath)
    if nargin >= 1 && isfile(filepath)
        catalog = readtable(filepath);
        catalog = rmmissing(catalog);
    else
        warning('loadLandslideCatalog:fileNotFound', ...
            'CSV not found at "%s" -- using synthetic placeholder catalog.', string(filepath));
        eventDate = datetime({'2023-07-12','2023-08-03','2024-06-21'})';
        location  = {'Sector A','Sector B','Sector A'}';
        severity  = {'Moderate','Severe','Minor'}';
        catalog = table(eventDate, location, severity, 'VariableNames', {'EventDate','Location','Severity'});
    end
end


%% ============================================================
% FUNCTION GENERATION FUNCTIONS (Sreenanda's part)
%% ============================================================

function FS = computeSlopeStability(slopeAngle_deg, cohesion_kPa, soilMoisture_pct)
    unitWeight = 18; % kN/m^3, typical soil unit weight
    soilDepth = 2;   % m, depth of potential failure plane
    phi = 30;        % degrees, internal friction angle

    beta = deg2rad(slopeAngle_deg);
    u = (soilMoisture_pct/100) .* unitWeight .* soilDepth .* 0.5; % pore pressure proxy

    numerator = cohesion_kPa + (unitWeight.*soilDepth.*cos(beta).^2 - u).*tand(phi);
    denominator = unitWeight.*soilDepth.*sin(beta).*cos(beta);

    FS = numerator ./ denominator;
    FS(denominator <= 0) = Inf;
end

function riskFactor = computeRainfallRiskFactor(rainfallTable, windowDays)
    if nargin < 2, windowDays = 3; end
    n = height(rainfallTable);
    cumulativeRain = zeros(n,1);
    for i = 1:n
        startIdx = max(1, i-windowDays+1);
        cumulativeRain(i) = sum(rainfallTable.Rainfall_mm(startIdx:i));
    end
    riskFactor = min(cumulativeRain / 150, 1); % >150mm in window = extreme (risk = 1)
end

function level = classifyRiskLevel(riskScore)
    level = strings(size(riskScore));
    level(riskScore < 30) = "Low";
    level(riskScore >= 30 & riskScore < 60) = "Moderate";
    level(riskScore >= 60 & riskScore < 80) = "High";
    level(riskScore >= 80) = "Severe";
end

function [riskScore, riskLevel] = computeLandslideRiskScore(FS, rainfallRiskFactor)
    instability = max(0, min(1, (2 - FS) / 1.5));
    combined = 0.6*instability + 0.4*rainfallRiskFactor;
    riskScore = round(combined * 100, 1);
    riskLevel = classifyRiskLevel(riskScore);
end
